```csharp
// ============================================================
// 蔚蓝风格角色控制器 - 完整伪代码
// 包含：可变重力、空中瞬间转向、跳跃缓冲、Coyote Time、冲刺
// ============================================================

class CelesteStyleController
{
// ---------- 物理状态（微分方程的状态变量） ----------
Vector2 position;
Vector2 velocity;
bool isGrounded;

// ---------- 可调参数（手感调教核心） ----------
// 地面参数
float groundAccel = 1200f;          // 地面加速力度（极大，接近瞬间满速）
float groundFriction = 0.8f;        // 地面减速系数
float maxGroundSpeed = 6.5f;

// 空中参数（破坏牛顿第二定律的地方）
float maxAirSpeed = 6.5f;           // 空中最大速度
float airAccel = 800f;              // 空中加速（极小，但转向靠直接赋值）

// 跳跃参数（破坏恒定重力）
float jumpSpeed = -9.5f;            // 起跳瞬间速度
float baseGravity = 800f;           // 基础重力加速度
float jumpGravityMult = 0.65f;      // 上升时重力倍率（飘）
float fallGravityMult = 1.2f;       // 下落时重力倍率（砸）
float releaseJumpMult = 3.0f;       // 松开按键时重力倍率（截断跳跃）

// ---------- 时间缓冲参数（破坏时间连续性） ----------
float jumpBufferTimer = 0f;
const float BUFFER_WINDOW = 0.12f;  // 120ms 缓冲窗口

float coyoteTimer = 0f;
const float COYOTE_TIME = 0.08f;    // 80ms 土狼时间（约5帧）

// ---------- 冲刺参数（破坏位置连续性） ----------
bool isDashing = false;
float dashTimer = 0f;
const float DASH_DURATION = 0.15f;
const float DASH_DISTANCE = 3.5f;
float dashCooldown = 0f;
const float DASH_COOLDOWN_TIME = 0.3f;

// ---------- 输入状态 ----------
float moveInput;        // -1 左, 0 不动, 1 右
bool jumpPressed;
bool jumpHeld;          // 按住跳跃键
bool dashPressed;
Vector2 dashDirection;

// ============================================================
// 主更新循环（每帧调用一次）
// ============================================================
void Update(float dt)
{
// ----- 1. 读取玩家输入 -----
ReadInput();

// ----- 2. 更新计时器（缓冲和土狼） -----
UpdateTimers(dt);

// ----- 3. 核心物理更新（破坏版） -----
UpdatePhysics(dt);

// ----- 4. 冲刺逻辑（独立于物理） -----
UpdateDash(dt);

// ----- 5. 碰撞检测与响应（地面判定） -----
ResolveCollisions();

// ----- 6. 最终位置更新 -----
position += velocity * dt;
}

// ============================================================
// 1. 输入读取
// ============================================================
void ReadInput()
{
// 左右移动（键盘或手柄摇杆）
moveInput = Input.GetAxis("Horizontal");

// 跳跃：按下瞬间触发，按住持续检测
jumpPressed = Input.GetButtonDown("Jump");
jumpHeld = Input.GetButton("Jump");

// 冲刺：按下瞬间触发
dashPressed = Input.GetButtonDown("Dash");
if (dashPressed)
{
// 获取冲刺方向（八方向）
float dashX = Input.GetAxis("DashHorizontal");  // 可单独设键
float dashY = Input.GetAxis("DashVertical");
dashDirection = new Vector2(dashX, dashY).Normalized();

// 如果没有方向输入，默认朝面向方向冲刺
if (dashDirection == Vector2.zero)
dashDirection = new Vector2(Mathf.Sign(velocity.x), 0);
}
}

// ============================================================
// 2. 更新计时器（破坏时间连续性的核心）
// ============================================================
void UpdateTimers(float dt)
{
// ----- 跳跃缓冲：玩家在空中按下跳跃，计时器开始倒计时 -----
if (jumpPressed)
jumpBufferTimer = BUFFER_WINDOW;
else if (jumpBufferTimer > 0)
jumpBufferTimer -= dt;

// ----- Coyote Time：离开地面后开始倒计时 -----
if (isGrounded)
coyoteTimer = COYOTE_TIME;
else if (coyoteTimer > 0)
coyoteTimer -= dt;

// ----- 冲刺冷却 -----
if (dashCooldown > 0)
dashCooldown -= dt;
}

// ============================================================
// 3. 核心物理更新（破坏微积分的地方）
// ============================================================
void UpdatePhysics(float dt)
{
// ----- 水平运动：破坏牛顿第二定律 -----
if (isGrounded)
{
// 地面：极大加速度，接近瞬间满速
if (moveInput != 0)
{
// 快速加速到目标速度（相当于硬赋值+微滑）
float targetSpeed = moveInput * maxGroundSpeed;
velocity.x += Mathf.Sign(targetSpeed - velocity.x) * groundAccel * dt;

// 如果已经很接近目标，直接赋值（消除微小的延迟）
if (Mathf.Abs(velocity.x - targetSpeed) < 0.5f)
velocity.x = targetSpeed;
}
else
{
// 松开按键：快速减速（有摩擦，但很快）
velocity.x *= (1f - groundFriction * dt);
if (Mathf.Abs(velocity.x) < 0.01f)
velocity.x = 0f;
}
}
else // 空中！！！
{
// ===== 核心破坏点：空中转向直接赋值 =====
// 不累加加速度，直接把输入映射到速度
if (moveInput != 0)
{
// 直接设置速度为输入方向 × 最大空中速度
// 这完全绕过了 F = ma 的微分过程
velocity.x = moveInput * maxAirSpeed;
}
else
{
// 松开按键：极快减速（不是完全瞬间，保留一点物理感）
velocity.x *= 0.9f;
if (Mathf.Abs(velocity.x) < 0.05f)
velocity.x = 0f;
}
}

// 水平速度限幅（防止数值溢出）
velocity.x = Mathf.Clamp(velocity.x, -maxGroundSpeed, maxGroundSpeed);

// ----- 垂直运动：破坏恒定重力（可变重力） -----
if (!isDashing)  // 冲刺时重力暂停
{
float gravity = baseGravity;

if (velocity.y < 0) // 上升阶段
{
// 上升：重力减小，跳跃更高更飘
gravity *= jumpGravityMult;

// 如果松开跳跃键，立即截断跳跃（破坏抛物线连续性）
if (!jumpHeld)
gravity *= releaseJumpMult;
}
else // 下落阶段
{
// 下落：重力增大，手感更脆
gravity *= fallGravityMult;
}

// 应用重力（这是唯一保留的微分积分）
velocity.y += gravity * dt;

// 最大下落速度限制（防止自由落体过快）
if (velocity.y > 20f)
velocity.y = 20f;
}

// ----- 跳跃执行（检查缓冲和土狼时间） -----
// 核心破坏点：允许在空中通过缓冲和土狼时间触发跳跃
if (jumpBufferTimer > 0 && (isGrounded || coyoteTimer > 0))
{
PerformJump();
jumpBufferTimer = 0f;  // 消耗缓冲
coyoteTimer = 0f;      // 消耗土狼时间
}
}

// ============================================================
// 跳跃执行
// ============================================================
void PerformJump()
{
// 起跳瞬间给一个向上的速度
velocity.y = jumpSpeed;

// 小技巧：起跳时稍微修正水平速度，让跳跃更可控
if (Mathf.Abs(velocity.x) < 0.5f)
velocity.x = moveInput * maxGroundSpeed * 0.5f;

// 重置土狼时间（防止连跳）
coyoteTimer = 0f;
}

// ============================================================
// 4. 冲刺逻辑（破坏位置连续性）
// ============================================================
void UpdateDash(float dt)
{
if (dashPressed && !isDashing && dashCooldown <= 0)
{
// 开始冲刺
isDashing = true;
dashTimer = DASH_DURATION;
dashCooldown = DASH_COOLDOWN_TIME;

// 破坏位置连续性：直接位移
position += dashDirection * DASH_DISTANCE;

// 破坏动量守恒：速度归零
velocity = Vector2.zero;

// 如果冲撞到墙壁，立即停止（防止穿模）
// （实际项目中这里会做碰撞检测）
}

if (isDashing)
{
dashTimer -= dt;
if (dashTimer <= 0)
{
isDashing = false;
}
}
}

// ============================================================
// 5. 碰撞检测与响应（简化版）
// ============================================================
void ResolveCollisions()
{
// 这里只做地面检测的简化版本
// 实际《蔚蓝》会做像素级精确的瓷砖碰撞

bool wasGrounded = isGrounded;
isGrounded = false;

// 检测脚下是否有地面（向下射线检测或瓷砖碰撞）
if (CheckGroundBelow(position))
{
isGrounded = true;

// 落地时修正位置（防止微小的穿透）
position.y = Mathf.Floor(position.y) + 1f; // 对齐到像素网格

// 落地时停止下落
if (velocity.y > 0)
velocity.y = 0f;
}

// 墙壁检测（简化）
// 如果撞墙且速度朝向墙壁，将水平速度置零
// ...
}

// ============================================================
// 辅助函数：地面检测（占位）
// ============================================================
bool CheckGroundBelow(Vector2 pos)
{
// 实际项目中这里会做射线检测或瓷砖碰撞
// 返回 true 表示脚下有地面
return Physics.CheckBox(pos + Vector2.down * 0.5f, new Vector2(0.4f, 0.1f), ...);
}
}
```

## 参数调教指南（手感配方）

你可以通过调整参数来获得不同的手感：

| 参数 | 调大效果 | 调小效果 | 《蔚蓝》参考值 |
| --- | --- | --- | --- |
| `groundAccel` | 地面加速更快（更灵敏） | 地面有迟滞感（像滑冰） | 1200 |
| `maxAirSpeed` | 空中移动更快 | 空中移动更慢 | 6.5 |
| `jumpSpeed` | 跳得更高 | 跳得更低 | -9.5 |
| `jumpGravityMult` | 跳跃更高、更飘 | 跳跃更低、更重 | 0.65 |
| `fallGravityMult` | 下落更快（更硬核） | 下落更慢（更休闲） | 1.2 |
| `releaseJumpMult` | 松开按键快速截断 | 松开按键影响小 | 3.0 |
| `BUFFER_WINDOW` | 更宽容的输入窗口 | 需要更精准的时机 | 0.12s |
| `COYOTE_TIME` | 更宽容的边缘判定 | 更严格的边缘判定 | 0.08s |

## 核心破坏点总结（对照代码）

| 破坏的物理法则 | 代码位置 | 效果 |
| --- | --- | --- |
| 牛顿第二定律（空中加速） | `UpdatePhysics()` 空中分支 | 空中转向直接赋值，无惯性 |
| 恒定重力 | `UpdatePhysics()` 重力计算 | 上升/下落/松开使用不同重力 |
| 时间连续性 | `UpdateTimers()` \+ 跳跃判定 | 缓冲窗口 \+ Coyote Time |
| 位置连续性 | `UpdateDash()` | 冲刺瞬移 \+ 速度归零 |
| 动量守恒 | `PerformJump()` \+ 冲刺 | 跳跃可截断，冲刺清空速度 |
